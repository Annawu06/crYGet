cmake_minimum_required(VERSION 3.20)
project(crYGet LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
if(NOT WIN32)
    find_package(PkgConfig REQUIRED)
    find_package(CURL REQUIRED)
    pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil)
    pkg_check_modules(UV REQUIRED IMPORTED_TARGET libuv)
    find_path(NODE_INCLUDE_DIR node.h PATH_SUFFIXES node REQUIRED)
    find_library(NODE_LIBRARY NAMES node REQUIRED)
    set(V8_INCLUDE_DIR "${NODE_INCLUDE_DIR}")
    set(V8_LIBRARIES "${NODE_LIBRARY}")
else()
    if(NOT FFMPEG_ROOT)
        message(FATAL_ERROR "Set FFMPEG_ROOT to an LGPL-compatible FFmpeg development build")
    endif()
    find_path(FFMPEG_INCLUDE_DIR libavformat/avformat.h HINTS "${FFMPEG_ROOT}/include" REQUIRED)
    foreach(component avformat avcodec avutil)
        find_library(FFMPEG_${component}_LIBRARY NAMES ${component} HINTS "${FFMPEG_ROOT}/lib" REQUIRED)
        list(APPEND FFMPEG_LIBRARIES "${FFMPEG_${component}_LIBRARY}")
    endforeach()
    if(NOT NODE_ROOT)
        message(FATAL_ERROR "Set NODE_ROOT to an embeddable Node/V8 SDK built for this Windows toolchain")
    endif()
    find_path(NODE_INCLUDE_DIR node.h HINTS "${NODE_ROOT}/include/node" "${NODE_ROOT}/include" REQUIRED)
    find_path(NODE_UV_INCLUDE_DIR uv.h HINTS "${NODE_ROOT}/include/node" "${NODE_ROOT}/include" REQUIRED)
    find_library(NODE_LIBRARY NAMES node HINTS "${NODE_ROOT}/lib" REQUIRED)
    find_library(NODE_UV_LIBRARY NAMES uv libuv HINTS "${NODE_ROOT}/lib" REQUIRED)
    set(V8_INCLUDE_DIR "${NODE_INCLUDE_DIR}")
    set(V8_LIBRARIES "${NODE_LIBRARY}" "${NODE_UV_LIBRARY}")
endif()

add_library(cryget-acorn STATIC third_party/acorn/acorn_data.c)
target_compile_definitions(cryget-acorn PRIVATE _GNU_SOURCE)

add_library(cryget-core STATIC src/video_core.cpp src/diagnostics.cpp src/player.cpp src/media_process.cpp)
target_include_directories(cryget-core PRIVATE "${V8_INCLUDE_DIR}" ${NODE_UV_INCLUDE_DIR})
target_link_libraries(cryget-core PUBLIC Threads::Threads cryget-acorn ${V8_LIBRARIES})
if(WIN32)
    target_compile_definitions(cryget-core PRIVATE BUILDING_NODE_EXTENSION)
endif()
if(WIN32)
    target_include_directories(cryget-core PRIVATE "${FFMPEG_INCLUDE_DIR}")
    target_link_libraries(cryget-core PUBLIC ${FFMPEG_LIBRARIES})
else()
    target_link_libraries(cryget-core PUBLIC PkgConfig::FFMPEG PkgConfig::UV)
endif()
if(WIN32)
    target_sources(cryget-core PRIVATE src/winhttp_download.cpp)
    target_link_libraries(cryget-core PUBLIC winhttp)
    target_compile_definitions(cryget-core PUBLIC NOMINMAX UNICODE _UNICODE)
else()
    target_link_libraries(cryget-core PUBLIC CURL::libcurl)
endif()
if(MSVC)
    add_compile_options(/utf-8)
    target_compile_options(cryget-core PRIVATE /utf-8)
endif()

enable_testing()
add_executable(cryget-core-test src/core_test.cpp)
target_link_libraries(cryget-core-test PRIVATE cryget-core)
add_test(NAME core-test COMMAND cryget-core-test)
add_executable(cryget-player-test src/player_test.cpp)
target_link_libraries(cryget-player-test PRIVATE cryget-core)
add_test(NAME player-test COMMAND cryget-player-test)
add_executable(cryget-mux-test src/mux_test.cpp)
target_link_libraries(cryget-mux-test PRIVATE cryget-core)
file(TO_CMAKE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures" CRYGET_FIXTURE_DIR)
configure_file(src/mux_test_config.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/mux_test_config.hpp" @ONLY)
target_include_directories(cryget-mux-test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME mux-test COMMAND cryget-mux-test)
if(WIN32)
    enable_language(RC)
    configure_file(src/windows.rc.in "${CMAKE_CURRENT_BINARY_DIR}/cryget.rc" @ONLY)
    set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/cryget.rc" PROPERTIES
        OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/cryget.ico")
    add_executable(cryget-desktop WIN32 src/windows.cpp "${CMAKE_CURRENT_BINARY_DIR}/cryget.rc")
    target_link_libraries(cryget-desktop PRIVATE cryget-core shell32 ole32 windowscodecs)
    if(MINGW)
        target_link_options(cryget-desktop PRIVATE -municode -static)
    endif()
else()
    pkg_check_modules(DESKTOP REQUIRED IMPORTED_TARGET x11 xft fontconfig libjpeg)
    add_executable(cryget-desktop src/main.cpp)
    target_link_libraries(cryget-desktop PRIVATE cryget-core PkgConfig::DESKTOP)
    add_test(NAME self-test COMMAND cryget-desktop --self-test)
endif()
